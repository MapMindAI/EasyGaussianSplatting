# JobParameters defaults shipped by the gsplat training server.
#
# The main detail lever is the final Gaussian count. Lower grow_grad2d or
# prune_opa, or raise refine_stop_iter, to retain more detail at the cost of
# GPU memory. DefaultStrategy has no upper bound on the count.
#
# floater_reg_weight is applied to both opacity_reg and scale_reg. It suppresses
# low-opacity and oversized Gaussians; lower it for more detail and higher GPU
# memory use.
#
# grow_grad2d is the image-plane gradient threshold for splitting or duplicating
# Gaussians. absgrad uses absolute rather than averaged gradients and therefore
# requires a higher threshold than the gsplat default.
#
# prune_opa removes Gaussians below the opacity threshold. grow_scale3d selects
# duplication below the 3D scale threshold; prune_scale3d removes larger ones.
# prune_scale2d removes Gaussians covering too much of an image while the
# refinement scale window is active.
#
# refine_start_iter, refine_stop_iter, and refine_every control densification.
# reset_every periodically resets opacity so unused Gaussians can be pruned.
# strategy selects between the two densification strategies. STRATEGY_DEFAULT
# has no ceiling on the Gaussian count, so on a fixed GPU it grows until
# training runs out of memory. STRATEGY_MCMC holds the count at cap_max,
# which is how to ask for a specific budget.
#
# This file is the schema's defaults; a file passed to --parameters is
# layered over it and only has to name what it changes. Its schema is
# gsplat.proto.

# Number of optimization steps and evaluation/save schedule.
iterations: 30000
# COLMAP image downsample factor used by the trainer.
data_factor: 1
# Regularization weight applied to both opacity and scale terms.
floater_reg_weight: 0.01

# Minimum opacity below which a Gaussian is pruned. It also sets the opacity
# reset floor, which gsplat takes as prune_opa * 2, so lowering it does not
# simply prune less: at 0.002 every reset clamps opacities to 0.004, and with
# floater_reg_weight 0.01 pushing them down they never recover above the prune
# threshold. That collapses this dataset from 409k Gaussians to 10k.
prune_opa: 0.005    # 0.005
# Image-plane gradient threshold for Gaussian splitting or duplication.
grow_grad2d: 0.0006   # 0.0006
# Use absolute image-plane gradients instead of averaged gradients, following the AbsGS paper.
absgrad: true
# 3D scale below which densification duplicates a Gaussian.
grow_scale3d: 0.01   # 0.01
# Maximum 3D scale; larger Gaussians are pruned.
prune_scale3d: 0.1  # 0.1
# Maximum image-plane scale before a Gaussian is pruned.
prune_scale2d: 0.15   # 0.15
# First iteration at which densification runs.
refine_start_iter: 500
# Last iteration at which densification runs.
refine_stop_iter: 20000
# Number of iterations between densification passes.
refine_every: 100
# Number of iterations between opacity resets.
reset_every: 3000


# Densification strategy: STRATEGY_DEFAULT or STRATEGY_MCMC.
# STRATEGY_DEFAULT: relying mainly on gradient-threshold-based split/clone/prune rules
# STRATEGY_MCMC tries to maintain a population of Gaussians while moving that population
# toward useful parts of the scene, by periodically:
# * teleporting low-opacity Gaussians to locations associated with high opacity,
# * adding new Gaussians according to the opacity distribution,
# * perturbing Gaussian positions with noise.
# In my experiments (mapping/benchmark) STRATEGY_MCMC out-performed STRATEGY_DEFAULT
strategy: STRATEGY_MCMC
# MCMC only: Gaussian count to converge to. Ignored by STRATEGY_DEFAULT.
cap_max: 500000


# Spherical-harmonic degree for view-dependent colour.
sh_degree: 3
# Weight of the SSIM term in the loss; the rest is L1. 0.5 beat gsplat's 0.2
# on every metric at no cost.
ssim_lambda: 0.5
# Packed rasterization: less GPU memory, slightly slower.
packed: true

# Gaussian learning rates. These match simple_trainer's defaults.
# Positions are scaled by the scene size internally, so keep means_lr small.
means_lr: 0.00016
scales_lr: 0.005
opacities_lr: 0.05
quats_lr: 0.001
# Degree-zero spherical harmonics control base colour; higher degrees control
# view-dependent detail and train more slowly.
sh0_lr: 0.0025
shN_lr: 0.000125

# Optimize camera poses alongside the model.
pose_opt: false
# Learning rate for camera-pose optimization. Only read when pose_opt is set.
pose_opt_lr: 0.00001
# Segment people out of the images before training, when the upload carried no
# masks/ of its own. Costs a few minutes and needs the segmentation weights.
run_segmentation: true
# Mask the sky along with the people. Only read when run_segmentation is set.
# Left on, the sky is masked out: it is at infinity, so the Gaussians that chase
# it are floaters that cost memory and blur the geometry below. It is off here
# because background_color now gives the sky something to render as.
mask_sky: false
# Colour the splats are composited over, in [0, 1] per channel. Leave it out
# and gsplat renders onto black instead. See doc/gsplat_server.md.
background_color { red: 1.0 green: 1.0 blue: 1.0 }


# TODO: depth loss still need more work. depth and render performance seems could
#       be obtain at the same time
# Predict depth with Depth Anything 3 and supervise training on it, when the
# upload carried no depths/ of its own. Off by default: it costs another pass
# over the images and needs the depth model served alongside the others.
run_depth: false
# Weight of the depth term in the loss. gsplat's own default, which is tuned
# for the sparse SfM depths rather than the denser ones generate_depth.py
# writes, so lower it if depth starts to dominate the photometric terms.
depth_lambda: 0.01
